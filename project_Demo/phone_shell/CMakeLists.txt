cmake_minimum_required(VERSION 3.16)
project(phone_shell C)
# The validated phone profile keeps its optimizations enabled by default.
option(YMGUI_OCCLUSION "Opaque coverage draw planner" ON)
option(YMGUI_CLIPBOARD_DYNAMIC "Allocate clipboard in memory region 1" ON)
include(${CMAKE_CURRENT_LIST_DIR}/../ymgui_app.cmake)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/phone_app_catalog.def")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/phone_app_catalog.def" phone_app_entries REGEX "^PHONE_APP\\(")
set(phone_app_sources)
foreach(entry IN LISTS phone_app_entries)
    if(NOT entry MATCHES "^PHONE_APP\\([A-Z_]+, *([a-z0-9_]+)\\)$")
        message(FATAL_ERROR "Invalid phone app entry: ${entry}")
    endif()
    list(APPEND phone_app_sources "apps/${CMAKE_MATCH_1}.c")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/phone_quick_catalog.def")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/phone_quick_catalog.def" phone_quick_entries REGEX "^PHONE_QUICK\\(")
set(phone_quick_sources)
foreach(entry IN LISTS phone_quick_entries)
    if(NOT entry MATCHES "^PHONE_QUICK\\([a-z0-9_]+, *([a-z0-9_]+)\\)$")
        message(FATAL_ERROR "Invalid quick setting entry: ${entry}")
    endif()
    list(APPEND phone_quick_sources "${CMAKE_MATCH_1}.c")
endforeach()
list(REMOVE_DUPLICATES phone_quick_sources)
ymgui_add_app(phone_shell phone_shell.c phone_font.c phone_locale.c phone_launcher.c
    phone_ui.c phone_app.c phone_shade.c phone_desktop.c phone_quick.c phone_quick_defaults.c ${phone_quick_sources} ${phone_app_sources})
target_include_directories(phone_shell PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(phone_shell PRIVATE GB2312_BIN_PATH="${YMGUI_REPO_ROOT}/tools/gb2312_glyphs.bin")
option(YAOMI_IME "Phone-wide Chinese software keyboard" ON)
target_compile_definitions(phone_shell PRIVATE YAOMI_IME=$<BOOL:${YAOMI_IME}>)
if(YAOMI_IME)
    target_sources(phone_shell PRIVATE phone_ime.c)
    target_include_directories(phone_shell PRIVATE ../chinese_ime)
    target_compile_definitions(phone_shell PRIVATE
        YMGUI_IME_CHARS_EXTERNAL=1 YMGUI_IME_PHRASES_EXTERNAL=1
        CHAR_BIN_PATH="${CMAKE_CURRENT_LIST_DIR}/../chinese_ime/pinyin_gb2312.bin"
        PHRASE_BIN_PATH="${CMAKE_CURRENT_LIST_DIR}/../chinese_ime/phrases_rime.bin")
endif()

if(YMGUI_PORTABLE_APP)
    target_compile_definitions(phone_shell PRIVATE PHONE_PORTABLE_BUILD=1)
    enable_testing()
    add_executable(phone_portable_test tests/test_portable.c)
    target_include_directories(phone_portable_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(phone_portable_test PRIVATE YAOMI_IME=$<BOOL:${YAOMI_IME}>)
    target_link_libraries(phone_portable_test PRIVATE phone_shell m)
    add_test(NAME phone_portable COMMAND phone_portable_test)
elseif(NOT ANDROID)
    enable_testing()
    add_test(NAME phone_shell_ui COMMAND phone_shell --selftest)
    set_tests_properties(phone_shell_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    add_executable(phone_app_registry_test phone_app.c phone_quick.c tests/test_registry.c)
    target_include_directories(phone_app_registry_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(phone_app_registry_test PRIVATE PHONE_APP_CATALOG="tests/catalog.def")
    target_link_libraries(phone_app_registry_test PRIVATE ymgui)
    add_test(NAME phone_app_registry COMMAND phone_app_registry_test)
endif()
