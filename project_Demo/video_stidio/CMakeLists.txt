cmake_minimum_required(VERSION 3.16)
project(video_stidio C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    message(FATAL_ERROR "video_stidio currently requires desktop Linux (POSIX media workers)")
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../ymgui_app.cmake)
set(CMAKE_C_STANDARD 11)
if(NOT YMGUI_COLOR_DEPTH STREQUAL "16")
    message(FATAL_ERROR "video_stidio currently uses RGB565; configure YMGUI_COLOR_DEPTH=16")
endif()
find_package(Threads REQUIRED)
include(cmake/ffmpeg.cmake)
add_library(studio_model STATIC model/project.c model/history.c model/render_plan.c)
target_include_directories(studio_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
find_program(STUDIO_PROXY_FFMPEG ffmpeg REQUIRED)
add_library(studio_media STATIC media/engine.c media/proxy.c media/decoder.c media/export.c)
target_include_directories(studio_media PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(studio_media PUBLIC studio_ffmpeg studio_model Threads::Threads)
target_compile_definitions(studio_media PRIVATE STUDIO_PROXY_FFMPEG="${STUDIO_PROXY_FFMPEG}")
add_library(studio_app STATIC app/studio.c app/actions.c app/bindings.c app/export_actions.c ui/workspace.c ui/timeline.c ui/thumbnails.c ui/dialogs.c)
target_include_directories(studio_app PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${YMGUI_SDL_INCLUDE_DIRS})
target_link_libraries(studio_app PUBLIC studio_model studio_media ymgui sdl_lcd m)
target_compile_definitions(studio_app PRIVATE GB2312_BIN_PATH="${YMGUI_REPO_ROOT}/tools/gb2312_glyphs.bin")
foreach(target studio_model studio_media studio_app)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
endforeach()
add_executable(video_stidio main.c)
target_link_libraries(video_stidio PRIVATE studio_app)
include(CTest)
if(BUILD_TESTING)
    add_executable(studio_model_test tests/model_test.c)
    target_link_libraries(studio_model_test PRIVATE studio_model)
    add_test(NAME studio_model COMMAND studio_model_test)
    add_executable(studio_media_test tests/media_test.c)
    target_link_libraries(studio_media_test PRIVATE studio_media)
    add_test(NAME studio_media COMMAND studio_media_test)
    set_tests_properties(studio_media PROPERTIES TIMEOUT 20)
    add_executable(studio_integration_test tests/integration_test.c)
    target_link_libraries(studio_integration_test PRIVATE studio_app)
    set(STUDIO_TEST_VIDEO "${CMAKE_CURRENT_BINARY_DIR}/integration-source.mp4")
    add_test(NAME studio_fixture COMMAND "${STUDIO_PROXY_FFMPEG}" -nostdin -v error
        -f lavfi -i "testsrc2=size=160x90:rate=25:duration=96"
        -c:v mpeg4 -g 25 -threads 1 -y "${STUDIO_TEST_VIDEO}")
    set_tests_properties(studio_fixture PROPERTIES FIXTURES_SETUP studio_video TIMEOUT 30)
    add_test(NAME studio_integration COMMAND studio_integration_test "${STUDIO_TEST_VIDEO}")
    set_tests_properties(studio_integration PROPERTIES FIXTURES_REQUIRED studio_video)
    add_executable(studio_export_test tests/export_test.c)
    target_link_libraries(studio_export_test PRIVATE studio_app)
    add_test(NAME studio_export COMMAND studio_export_test)
    set_tests_properties(studio_export PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 90)
    foreach(test studio_model_test studio_media_test studio_integration_test studio_export_test)
        target_compile_options(${test} PRIVATE -UNDEBUG)
        target_compile_definitions(${test} PRIVATE STUDIO_PROXY_FFMPEG="${STUDIO_PROXY_FFMPEG}")
    endforeach()
    set_tests_properties(studio_integration PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 90)
endif()
