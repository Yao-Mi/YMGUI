cmake_minimum_required(VERSION 3.16)
project(music_studio C)
include(${CMAKE_CURRENT_LIST_DIR}/../ymgui_app.cmake)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "music_studio currently supports desktop Linux")
endif()
if(NOT YMGUI_COLOR_DEPTH MATCHES "^(16|24)$")
    message(FATAL_ERROR "music_studio requires RGB565 or RGB888")
endif()
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${YMGUI_SDL_INCLUDE_DIRS})
target_compile_definitions(ymgui PUBLIC GY_LABEL_TEXT_MAX=256)
add_library(music_engine STATIC model/project.c model/storage.c audio/audio.c audio/sample_bank.c)
set(MS_SAMPLE_DIR "${YMGUI_REPO_ROOT}/extern_lib/GeneralUser-GS")
set(MS_BODY_DIR "${YMGUI_REPO_ROOT}/extern_lib/BodyPercussion")
foreach(MS_DIR IN ITEMS "${MS_SAMPLE_DIR}" "${MS_BODY_DIR}")
    file(STRINGS "${MS_DIR}/samples.sha256" MS_HASH_ROWS)
    foreach(MS_ROW IN LISTS MS_HASH_ROWS)
        string(REGEX MATCH "^([0-9a-f]+)  ([a-z_]+[.][a-z0-9]+)$" MS_VALID_ROW "${MS_ROW}")
        if(NOT MS_VALID_ROW)
            message(FATAL_ERROR "Invalid sample manifest row: ${MS_ROW}")
        endif()
        set(MS_HASH "${CMAKE_MATCH_1}")
        set(MS_FILE "${CMAKE_MATCH_2}")
        file(SHA256 "${MS_DIR}/${MS_FILE}" MS_ACTUAL_HASH)
        if(NOT MS_ACTUAL_HASH STREQUAL MS_HASH)
            message(FATAL_ERROR "Sample checksum mismatch: ${MS_FILE}")
        endif()
    endforeach()
endforeach()
target_compile_definitions(music_engine PRIVATE MS_SAMPLE_DIR="${MS_SAMPLE_DIR}" MS_BODY_DIR="${MS_BODY_DIR}")
target_link_libraries(music_engine PUBLIC ${YMGUI_SDL_LIBRARIES} m)
target_compile_options(music_engine PRIVATE -Wall -Wextra -Wpedantic)
ymgui_add_app(music_studio main.c app/actions.c ui/workspace.c ui/timeline.c ui/editors.c ui/dialogs.c tests/ui_test.c)
target_link_libraries(music_studio music_engine)
target_compile_options(music_studio PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(music_studio PRIVATE
    GB2312_BIN_PATH="${YMGUI_REPO_ROOT}/tools/gb2312_glyphs.bin"
    MS_DATA_DIR="${CMAKE_CURRENT_BINARY_DIR}/user")
enable_testing()
add_executable(music_engine_test tests/engine_test.c)
target_link_libraries(music_engine_test music_engine)
target_compile_options(music_engine_test PRIVATE -UNDEBUG)
target_compile_definitions(music_engine_test PRIVATE MS_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME music_engine COMMAND music_engine_test ${CMAKE_CURRENT_BINARY_DIR})
add_test(NAME music_ui COMMAND music_studio --selftest)
set_tests_properties(music_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)

# 可选小样生成器：产出普通工程和 WAV，不改变默认启动工程。
add_executable(music_epic_demo EXCLUDE_FROM_ALL examples/epic_demo.c)
target_link_libraries(music_epic_demo music_engine)

add_executable(music_sound_demo EXCLUDE_FROM_ALL examples/sound_demo.c)
target_link_libraries(music_sound_demo music_engine)
