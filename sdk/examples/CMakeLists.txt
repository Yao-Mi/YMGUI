cmake_minimum_required(VERSION 3.16)
project(YMGUI_Binary_Examples C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT YMGUI_DIR)
    set(YMGUI_DIR "${CMAKE_CURRENT_LIST_DIR}/../cmake")
endif()
find_package(YMGUI CONFIG REQUIRED)
enable_testing()
add_executable(demo_sdk_minimal demo_sdk_minimal.c)
target_link_libraries(demo_sdk_minimal PRIVATE YMGUI::ymgui)
add_test(NAME sdk_minimal COMMAND demo_sdk_minimal)
if(TARGET YMGUI::sdl)
    add_executable(demo_sdk_window demo_sdk_window.c)
    target_link_libraries(demo_sdk_window PRIVATE YMGUI::sdl)
    add_test(NAME sdk_window COMMAND demo_sdk_window 30)
    set_tests_properties(sdk_window PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 15)
endif()
