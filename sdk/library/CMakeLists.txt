cmake_minimum_required(VERSION 3.16)
project(YMGUI_SDK_Libraries C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "The SDK packager currently builds native Linux libraries only")
endif()
set(YMGUI_COLOR_DEPTH 16 CACHE STRING "SDK framebuffer depth")
if(NOT YMGUI_COLOR_DEPTH MATCHES "^(16|24)$")
    message(FATAL_ERROR "SDK variants are 16 (RGB565) and 24 (RGB888)")
endif()
option(YMGUI_SDK_BUILD_SDL "Build the optional SDL adapter" ON)
get_filename_component(YMGUI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../YMGUI" ABSOLUTE)
file(GLOB_RECURSE _sources CONFIGURE_DEPENDS "${YMGUI_ROOT}/*.c")
file(GLOB_RECURSE _headers CONFIGURE_DEPENDS "${YMGUI_ROOT}/*.h")
set(_includes)
foreach(_header IN LISTS _headers)
    get_filename_component(_dir "${_header}" DIRECTORY)
    list(APPEND _includes "${_dir}")
endforeach()
list(REMOVE_DUPLICATES _includes)
add_library(ymgui STATIC ${_sources})
target_include_directories(ymgui PUBLIC ${_includes})
target_compile_definitions(ymgui PUBLIC YMGUI_COLOR_DEPTH=${YMGUI_COLOR_DEPTH})
target_link_libraries(ymgui PUBLIC m ${CMAKE_DL_LIBS})
set_target_properties(ymgui PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(YMGUI_SDK_BUILD_SDL)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    add_library(ymgui_sdl STATIC ../../SDL_LCD/SDL_LCD.c)
    target_include_directories(ymgui_sdl PUBLIC ../../SDL_LCD)
    target_link_libraries(ymgui_sdl PUBLIC ymgui PkgConfig::SDL2)
    set_target_properties(ymgui_sdl PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/platform.cmake"
    "set(YMGUI_SDK_SYSTEM \"${CMAKE_SYSTEM_NAME}\")\nset(YMGUI_SDK_PROCESSOR \"${CMAKE_SYSTEM_PROCESSOR}\")\nset(YMGUI_SDK_POINTER_SIZE ${CMAKE_SIZEOF_VOID_P})\n")
